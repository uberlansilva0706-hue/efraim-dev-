let pacoteSelecionado = "";

// Função para abrir o modal de solicitação
function abrirModal(pacote) {
  pacoteSelecionado = pacote;
  const tituloElemento = document.getElementById("modal-titulo");
  const modalElemento = document.getElementById("modal-checkout");
  
  if (tituloElemento && modalElemento) {
    tituloElemento.innerText = "Solicitar: " + pacote;
    modalElemento.style.display = "flex";
    modalElemento.setAttribute("aria-hidden", "false");
  }
}

// Função para fechar o modal
function fecharModal() {
  const modalElemento = document.getElementById("modal-checkout");
  if (modalElemento) {
    modalElemento.style.display = "none";
    modalElemento.setAttribute("aria-hidden", "true");
  }
}

// Envio do formulário com disparo formatado para o WhatsApp
function enviarSolicitacao(e) {
  e.preventDefault();
  
  const nome = document.getElementById("nome").value.trim();
  const email = document.getElementById("email").value.trim();
  const whatsapp = document.getElementById("whatsapp").value.trim();

  // Número corporativo de destino
  const numeroDestino = "5511999999999"; 
  
  const mensagem = `Olá! Meu nome é ${nome}.\nE-mail: ${email}\nTelefone: ${whatsapp}\n\nTenho interesse no pacote: *${pacoteSelecionado}*.\nGostaria de alinhar o escopo e receber um orçamento.`;
  
  const urlWhatsapp = `https://wa.me/${numeroDestino}?text=${encodeURIComponent(mensagem)}`;
  
  window.open(urlWhatsapp, "_blank");
  fecharModal();
}

// Alerta temporário para os botões do portfólio enquanto os links de testes são vinculados
function alertaDemonstracao(event, tipo) {
  event.preventDefault();
  alert(`O ambiente demonstrativo de [${tipo}] está ativo. Em breve este link direcionará para a instância em tempo real.`);
}

// Fechar modal ao clicar fora da caixa de conteúdo
window.onclick = function(event) {
  const modalElemento = document.getElementById("modal-checkout");
  if (event.target === modalElemento) {
    fecharModal();
  }
};
